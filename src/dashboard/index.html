  "type":       "sendEmail",
  "payload":    { "to": "u@x.com" },
  "priority":   5,
  "maxRetries": 3

Kubernetes Scaling

# Scale manually
kubectl scale deployment jq-worker-email --replicas=10 -n job-queue
# HPA auto-scale range
jq-worker-general  2 → 20 pods (CPU target 70%)
jq-worker-email   2 → 30 pods (CPU target 65%)
jq-worker-payment  2 → 10 pods (CPU target 60%)
# Stream worker logs
kubectl logs -f -l role=email -n job-queue
